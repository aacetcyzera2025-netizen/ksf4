#include <stdio.h>
#include <string.h>
#include <openssl/evp.h>

int main()
{
    char text[] = "hello world";
    unsigned char key[] = "12345678";
    unsigned char encrypted[100];

    int len, encrypted_len;

    EVP_CIPHER_CTX *ctx = EVP_CIPHER_CTX_new();

    if (ctx == NULL)
    {
        printf("Context creation failed\n");
        return 1;
    }

    if (EVP_EncryptInit_ex(ctx, EVP_bf_ecb(), NULL, key, NULL) != 1)
    {
        printf("Encryption initialization failed\n");
        EVP_CIPHER_CTX_free(ctx);
        return 1;
    }

    if (EVP_EncryptUpdate(ctx, encrypted, &len,
                          (unsigned char *)text, strlen(text)) != 1)
    {
        printf("Encryption failed\n");
        EVP_CIPHER_CTX_free(ctx);
        return 1;
    }

    encrypted_len = len;

    if (EVP_EncryptFinal_ex(ctx, encrypted + encrypted_len, &len) != 1)
    {
        printf("Final encryption failed\n");
        EVP_CIPHER_CTX_free(ctx);
        return 1;
    }

    encrypted_len += len;

    printf("Original Text: %s\n", text);
    printf("Encrypted Text: ");

    for (int i = 0; i < encrypted_len; i++)
    {
        printf("%02X", encrypted[i]);
    }

    printf("\n");

    EVP_CIPHER_CTX_free(ctx);

    return 0;
}
